#### 1.malloc()
int* p = malloc(5 * sizeof(int));
```c
向系统申请能够存放 5 个 int 的空间。
申请成功之后，会返回：这块内存空间的首地址
       malloc申请的空间
   ┌────┬────┬────┬────┬────┐
   │ 10 │ 20 │    │    │    │
   └────┴────┴────┴────┴────┘
     ↑
     p

*p = 10;//内存首个元素
p++ 和 *(p + i) 仍旧可以使用
```

以前：
```text
int arr[5];//数组空间已经存在  
int* p = arr;    
使用
```

现在：
```text
malloc() -> 申请空间 -> 返回地址 -> int* p -> 使用
```

```c
#include <stdio.h>
#include <stdlib.h>

int main()
{
    int* p = malloc(5 * sizeof(int));

    p[0] = 10;
    p[1] = 20;
    p[2] = 30;
    p[3] = 40;
    p[4] = 50;

    for (int i = 0; i < 5; i++) {
        printf("%d ", p[i]);//结果10 20 30 40 50
    }

    free(p);
    return 0;
}
```

malloc必须检查是否成功

如果成功，返回有效地址；

如果申请失败：malloc() 会返回NULL（即p==NULL）。

标准写法：
```c
int* p = malloc(5 * sizeof(int));

if (p == NULL) {
    printf("内存申请失败\n");
    return 1;
}
然后才使用：
p[0] = 10;
```


#### 2.free()
malloc -> 申请-> 使用 -> 释放

函数结束 -> p 消失 -> 但是 malloc 的内存还在

于是这块内存就失去了可以访问它的指针。

这就是：内存泄漏（Memory Leak）
```text
int* p = malloc(5 * sizeof(int));
free(p);//此时堆空间已经释放
```
但是 p这个指针变量本身还存在,它里面可能还保存着原来的地址！

这就产生了一个非常重要的问题：p 现在指向的空间已经不存在了！

这种指针叫：悬空指针（跑空了）

```c
int* p = malloc(...);
if (p != NULL)
{
    // 才使用
}
free(p);//释放空间
p=NULL;//不再让 p 指向已经释放的空间
```

#### 3.calloc()
它也是申请5个int,

但和 malloc() 有一个重要区别：calloc() 申请的空间会被初始化为 0!!!

int* p = calloc(5, sizeof(int));//格式


#### 4.realloc()
调整已经申请的动态内存大小。

int* p = malloc(5 * sizeof(int));

现在只有 5 个 int,如果想变成 10 个：

p = realloc(p, 10 * sizeof(int));

这里有一个非常重要的安全写法，
```c
int* temp = realloc(p, 10 * sizeof(int));
if (temp != NULL)
{
    p = temp;
}
```
```text
malloc
 ↓
申请5个int
 ↓
arr
 ↓
输入5个数据
 ↓
realloc
 ↓
尝试变成10个int
 ↓
 ┌───────────────┐
 │               │
成功             失败
 ↓               ↓
temp             temp == NULL
 ↓               ↓
arr = temp       arr仍然有效
 ↓               ↓
输入新增数据     free(arr)
 ↓
输出
 ↓
free(arr)
```

### 综合练习
#### 1.动态数组扩容

要求：
```text
1. 先申请5个int空间（判断是否申请成功）
2. 输入5个整数
3. 输出这5个整数
4. 使用realloc扩容到10个
5. 输入新增的5个整数
6. 输出全部10个整数
7. free释放内存
```
```c
#define _CRT_SECURE_NO_WARNINGS

#include<stdio.h>
#include<stdlib.h>

void practice() {
	//1.申请5个元素
	int* p = malloc(5 * sizeof(int));
	if (p == NULL) {
		printf("动态数组申请失败\n");
		return;
	}
	//2.输入5个元素
	for (int i = 0; i < 5; i++) {
		scanf("%d", p + i);
	}
	//3.输出这5个元素
	for (int i = 0; i < 5; i++) {
		printf("%d ", *(p + i));
	}
	//4.扩充到10个元素
	int* temp = realloc(p, 10 * sizeof(int));
	if (temp == NULL) {
		printf("内存申请失败\n");
		free(p);
		return;
	}
	p = temp;
	//5.输入新添加的元素
	printf("输入新添的元素：\n");
	for (int i = 5; i < 10; i++) {
		scanf("%d", p + i);
	}
	//6.输出新增后的数组
	printf("新增后的数组：\n");
	for (int i = 0; i < 10; i++) {
		printf("%d ", *(p + i));
	}
	
	free(p);
	p = NULL;
}

int main() {
	practice();
}
```

#### 2.动态扩容不限次数
这次不再规定：5 → 10

而是让程序自己判断：
```text
容量 = 5
↓
输入第1~5个
↓
第6个来了，空间不够，realloc扩容（temp）
↓
继续输入
↓
空间再次不足，再次realloc
```
一个思路：
```text
int capacity = 5;  // 当前容量
int size = 0;      // 当前实际元素数量

每输入一个元素：（设置普通变量value存值）
如果 size == capacity
    ↓
 先扩容
    ↓
 保存元素
    ↓
  size++
```

```c
#define _CRT_SECURE_NO_WARNINGS

#include<stdio.h>
#include<stdlib.h>

void practice2() {
	int size = 0;
	int capacity = 5;
	int* p = malloc(capacity * sizeof(int));
	if (p == NULL) {
		printf("动态内存申请失败\n");
		return;
	}
	while (1) {
		int value;//设置普通变量
		scanf("%d", &value);
		if (value == -1) {
			break;//结束条件：输入-1
		}
		
		if (size == capacity) {
			capacity *= 2;
			int*temp=realloc(p, capacity*sizeof(int));
			if (temp == NULL) {
				printf("扩容失败\n");
				free(p);
				return;
			}
			p = temp;
		}
		*(p + size) = value;
		size++;
	}
	printf("最终数组：");
	for (int i = 0; i < size; i++)
	{
		printf("%d ", *(p + i));
	}

	printf("\n");
	free(p);
	p = NULL;
}

int main() {
	practice2();
}
```
#### 3.内存泄漏 + 动态数组综合检查
实现：(建议三个函数)
```text
初始容量为 5
使用 malloc() 创建动态数组
不断输入整数
输入 -1 时结束
空间不足时使用 realloc()，容量扩大 2 倍
输出最终数组
写一个函数求最大值
写一个函数进行升序排序
输出最大值和排序后的数组
最后 free() 释放内存
```

```c
#define _CRT_SECURE_NO_WARNINGS

#include<stdio.h>
#include<stdlib.h>

void practice3() {
	int capacity = 5;
	int size = 0;
	int* p = malloc(capacity * sizeof(int));
	if (p == NULL) {
		printf("动态内存分配失败\n");
		return;
	}
	while (1) {
		int value;
		scanf("%d", &value);
		if (value == -1) {
			break;
		}
		if (size==capacity) {
			capacity *= 2;
			int* temp = realloc(p, capacity * sizeof(int));
			if (temp == NULL) {
				printf("扩容失败\n");
				free(p);
				return;
			}
			p = temp;
		}
		*(p + size)=value;//不用scanf,直接赋值
		size++;
	}
	for (int i = 0; i < size; i++) {
		printf("%d", *(p + i));
	}
	printf("\n");
	free(p);
	p = NULL;
}

void dynamic_max() {
	int size = 5;
	int* p = malloc(size * sizeof(int));
	if (p == NULL) {
		printf("动态内存分配失败\n");
		return;
	}
	for (int i = 0; i < size; i++) {
		scanf("%d", p + i);
	}
	
	int max = *(p);
	for (int i = 0; i < size; i++) {
		if (*(p + i) > max) {
			max = *(p + i);
		}
	}
	printf("%d", max);
	free(p);
	p = NULL;
}

void dynamic_sort() {
	int size = 5;
	int* p = malloc(size * sizeof(int));
	if (p == NULL) {
		printf("动态内存申请失败\n");
		return;
	}
	for (int i = 0; i < size; i++) {
		scanf("%d", p + i);
	}
	for (int i = 0; i < size; i++) {
		for (int j = 0; j < size - i-1; j++) {
			if (*(p + j) > *(p + j + 1)) {
				int temp = *(p + j);
				*(p + j) = *(p + j + 1);
				*(p + j + 1) = temp;
			}
		}
	}
	for (int i = 0; i < size; i++) {
		printf("%d", *(p + i));
	}
	free(p);
	p = NULL;
}

int main() {
	practice3();
	dynamic_max();
	dynamic_sort();
}
```
